IDE    = {Letter}{AlphaNumeric}*

! The grammar starts below
<Program> ::= <VariablesBlock>          
<VariablesBlock> ::=  variables '{' <Variables> '}'
<Variables> ::= <Variable> | <Variable> <Variables>
<Variable> ::= <Type> <IDE> ';' | <Type> <IDE> ',' <MultipleVariables> ';' 
<MultipleVariables> ::= <IDE> |  <IDE> ',' <MultipleVariables>
<Type> ::= int | real | boolean | string
<IDE> ::= IDE
